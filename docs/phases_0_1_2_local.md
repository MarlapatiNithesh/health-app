# MediBot — Phases 0 · 1 · 2 : Local Execution Guide
### Node.js + TypeScript · Local-first, no Docker, no CI/CD

> **Scope:** This document covers only **Phases 0, 1, and 2** — the local foundation you need before any AI or cloud work. No Docker, no CI/CD, no cloud services until the local version is solid.

---

## Prerequisites — Install Once

Before starting, ensure the following are installed locally:

| Tool | Version | Install |
|---|---|---|
| Node.js | 20 LTS | [nodejs.org](https://nodejs.org) |
| npm | 10+ | Bundled with Node.js |
| PostgreSQL | 16+ | [postgresql.org](https://www.postgresql.org/download/windows/) |
| Python | 3.10+ | [python.org](https://www.python.org) (for Phase 2 NER service) |
| Git | Latest | [git-scm.com](https://git-scm.com) |

> **Windows tip:** Use `winget install PostgreSQL.PostgreSQL` or install via the PostgreSQL installer. Make sure `psql` is in your PATH.

`apps/api-gateway/src/middleware/auth.ts`:
```typescript
import { Request, Response, NextFunction } from 'express';
import jwt from 'jsonwebtoken';
import type { JwtPayload } from '@medibot/types';

export function authMiddleware(req: Request, res: Response, next: NextFunction) {
  const header = req.headers.authorization;
  if (!header?.startsWith('Bearer ')) {
    return res.status(401).json({ error: 'Missing or invalid Authorization header' });
  }
  try {
    const token = header.split(' ')[1];
    const payload = jwt.verify(token, process.env.JWT_SECRET!) as JwtPayload;
    (req as any).user = payload;
    next();
  } catch {
    return res.status(401).json({ error: 'Token expired or invalid' });
  }
}
```

`apps/api-gateway/src/middleware/consent.ts`:
```typescript
import { Request, Response, NextFunction } from 'express';
import { prisma } from '@medibot/db';

export async function consentMiddleware(req: Request, res: Response, next: NextFunction) {
  const userId = (req as any).user?.userId;
  if (!userId) return res.status(401).json({ error: 'Unauthorized' });

  const consent = await prisma.consent.findFirst({
    where: { userId, scope: 'data_access', granted: true, revokedAt: null },
  });

  if (!consent) {
    return res.status(403).json({ error: 'Patient consent not granted for data access' });
  }
  next();
}
```

`apps/api-gateway/src/middleware/audit.ts`:
```typescript
import { Request, Response, NextFunction } from 'express';
import { prisma } from '@medibot/db';

export function auditMiddleware(req: Request, res: Response, next: NextFunction) {
  const originalEnd = res.end.bind(res);
  res.end = function (...args: any[]) {
    const userId = (req as any).user?.userId ?? null;
    prisma.auditLog.create({
      data: {
        userId,
        action: req.method,
        resource: req.path,
        ipAddress: req.ip,
        userAgent: req.get('user-agent'),
        statusCode: res.statusCode,
      },
    }).catch(console.error);
    return originalEnd(...args);
  };
  next();
}
```

## Step 1.3 — Patient Service Routes

**`apps/patient-service/src/routes/patient.ts`** (sample):
```typescript
import { Router, Request, Response } from 'express';
import { prisma } from '@medibot/db';
import { CreatePatientSchema } from '@medibot/types';

export const patientRouter = Router();

// POST /:userId/profile — Create patient profile
patientRouter.post('/:userId/profile', async (req: Request, res: Response) => {
  const parsed = CreatePatientSchema.safeParse(req.body);
  if (!parsed.success) return res.status(400).json({ errors: parsed.error.flatten() });

  try {
    const profile = await prisma.patientProfile.create({
      data: { userId: req.params.userId, ...parsed.data },
    });
    res.status(201).json(profile);
  } catch (e: any) {
    if (e.code === 'P2002') return res.status(409).json({ error: 'Profile already exists for this user' });
    res.status(500).json({ error: 'Internal server error' });
  }
});

// GET /:patientId — Get full Patient 360
patientRouter.get('/:patientId', async (req: Request, res: Response) => {
  const patient = await prisma.patientProfile.findUnique({
    where: { patientId: req.params.patientId },
    include: {
      conditions: { where: { deletedAt: null } },
      allergies:  { where: { deletedAt: null } },
      medications:{ where: { deletedAt: null, status: 'active' } },
      labResults: { orderBy: { observedAt: 'desc' }, take: 20 },
    },
  });
  if (!patient) return res.status(404).json({ error: 'Patient not found' });
  res.json(patient);
});
```

## Step 1.4 — Full API Endpoints (implement all of these)

```
POST   /patients/:userId/profile         – Create patient profile (links to User)
GET    /patients/:patientId              – Get Patient 360 (conditions, meds, allergies, recent labs)
PUT    /patients/:patientId/profile      – Update demographics / preferences

GET    /patients/:patientId/conditions   – List active medical conditions
POST   /patients/:patientId/conditions   – Add condition (with optional Provenance)
DELETE /patients/:patientId/conditions/:id – Soft-delete (set deletedAt)

GET    /patients/:patientId/allergies    – List allergies
POST   /patients/:patientId/allergies    – Add allergy
DELETE /patients/:patientId/allergies/:id

GET    /patients/:patientId/medications  – List active medications
POST   /patients/:patientId/medications  – Add medication
PUT    /patients/:patientId/medications/:id/stop – Stop medication (set status=stopped, stoppedAt)
```

### Provenance rule — enforced at service layer:
Every `POST` that writes a clinical record (condition, allergy, medication, lab) **must** create a `Provenance` record first, then attach `provenanceId` to the clinical record. Do not write clinical records without provenance:

```typescript
// Always create provenance before writing clinical record
async function withProvenance<T>(
  source: { sourceType: string; sourceDate: Date; documentId?: string; confidence?: number },
  writer: (provenanceId: string) => Promise<T>
): Promise<T> {
  const provenance = await prisma.provenance.create({
    data: { ...source, sourceDate: source.sourceDate },
  });
  return writer(provenance.sourceId);
}

// Usage:
const result = await withProvenance(
  { sourceType: 'manual_entry', sourceDate: new Date() },
  (provenanceId) => prisma.labResult.create({ data: { ...labData, provenanceId } })
);
```

## Step 2.3 — File Upload Endpoint

`apps/document-service/src/routes/documents.ts`:
```typescript
import { Router, Request, Response } from 'express';
import multer from 'multer';
import path from 'path';
import { v4 as uuid } from 'uuid';
import { prisma } from '@medibot/db';
import { extractionQueue } from '../jobs/queue';

const ALLOWED_TYPES = ['application/pdf', 'image/png', 'image/jpeg', 'image/tiff'];
const MAX_SIZE = parseInt(process.env.MAX_FILE_SIZE_MB ?? '20') * 1024 * 1024;

const storage = multer.diskStorage({
  destination: process.env.UPLOAD_DIR ?? './uploads',
  filename: (_, file, cb) => {
    // Never store original PHI-bearing filename
    const safeFilename = `${uuid()}${path.extname(file.originalname)}`;
    cb(null, safeFilename);
  },
});

const upload = multer({
  storage,
  limits: { fileSize: MAX_SIZE },
  fileFilter: (_, file, cb) => {
    if (ALLOWED_TYPES.includes(file.mimetype)) cb(null, true);
    else cb(new Error('Unsupported file type'));
  },
});

export const documentRouter = Router();

// POST /documents — Upload and queue for extraction
documentRouter.post('/', upload.single('file'), async (req: Request, res: Response) => {
  const { patientId, documentType } = req.body;
  if (!req.file) return res.status(400).json({ error: 'No file uploaded' });
  if (!patientId) return res.status(400).json({ error: 'patientId is required' });

  // Create DB record immediately
  const doc = await prisma.healthDocument.create({
    data: {
      patientId,
      type: documentType ?? 'other',
      originalFilename: req.file.originalname, // stored for display but NOT in URI
      storageUri: req.file.path,               // local path (will become S3 key in prod)
      mimeType: req.file.mimetype,
      extractionStatus: 'pending',
    },
  });

  // Enqueue background extraction job
  await extractionQueue.add('extract', { documentId: doc.documentId, filePath: req.file.path });

  res.status(202).json({ documentId: doc.documentId, status: 'pending' });
});

// GET /documents/:id/status
documentRouter.get('/:id/status', async (req: Request, res: Response) => {
  const doc = await prisma.healthDocument.findUnique({
    where: { documentId: req.params.id },
    select: { documentId: true, extractionStatus: true, type: true, uploadedAt: true },
  });
  if (!doc) return res.status(404).json({ error: 'Document not found' });
  res.json(doc);
});

// GET /documents/:id/extracted — Fields pending user confirmation
documentRouter.get('/:id/extracted', async (req: Request, res: Response) => {
  const fields = await prisma.extractedField.findMany({
    where: { documentId: req.params.id },
    include: { provenance: true },
  });
  res.json({ fields });
});

// POST /documents/:id/confirm — User confirms or rejects extracted fields
documentRouter.post('/:id/confirm', async (req: Request, res: Response) => {
  const { confirmations } = req.body as { confirmations: Array<{ fieldId: string; accepted: boolean; correctedValue?: string }> };

  await Promise.all(confirmations.map(async ({ fieldId, accepted, correctedValue }) => {
    await prisma.extractedField.update({
      where: { fieldId },
      data: {
        status: accepted ? 'confirmed' : 'rejected',
        normalizedValue: correctedValue ?? undefined,
        confirmedAt: new Date(),
      },
    });
  }));

  // Write confirmed fields to Patient 360
  const { commitConfirmedFields } = await import('../services/commit');
  await commitConfirmedFields(req.params.id);

  res.json({ message: 'Confirmations processed' });
});
```

## Step 2.5 — Extraction Pipeline Service

`apps/document-service/src/services/extraction.ts`:
```typescript
import { prisma } from '@medibot/db';
import { runOCR } from './ocr';
import { runNER } from './ner';
import { normalizeEntities } from './normalize';
import { scoreConfidence } from './confidence';
import { autoCommitHighConfidence } from './commit';

export async function runExtractionPipeline(documentId: string, filePath: string) {
  await prisma.healthDocument.update({
    where: { documentId },
    data: { extractionStatus: 'processing' },
  });

  try {
    // Step 1: OCR → raw text
    const rawText = await runOCR(filePath);

    // Step 2: NER → medical entities from Python microservice
    const entities = await runNER(rawText);

    // Step 3: Normalize → RxNorm, LOINC, ICD codes
    const normalized = await normalizeEntities(entities);

    // Step 4: Score confidence per field
    const scored = scoreConfidence(normalized);

    // Step 5: Create Provenance record for this extraction
    const provenance = await prisma.provenance.create({
      data: {
        sourceType: 'document',
        sourceDate: new Date(),
        documentId,
        extractionModel: 'scispacy-v3',
        modelVersion: '1.0.0',
        confidence: scored.avgConfidence,
      },
    });

    // Step 6: Write ExtractedField records
    await Promise.all(scored.fields.map((field) =>
      prisma.extractedField.create({
        data: {
          documentId,
          fieldName: field.name,
          rawValue: field.rawValue,
          normalizedValue: field.normalizedValue,
          confidence: field.confidence,
          status: field.confidence >= 0.85 ? 'confirmed' : 'pending',
          provenanceId: provenance.sourceId,
        },
      })
    ));

    // Step 7: Auto-commit HIGH confidence fields to Patient 360
    await autoCommitHighConfidence(documentId);

    await prisma.healthDocument.update({
      where: { documentId },
      data: { extractionStatus: 'done' },
    });

  } catch (err) {
    await prisma.healthDocument.update({
      where: { documentId },
      data: { extractionStatus: 'failed' },
    });
    throw err;
  }
}
```

## Step 2.6 — OCR Service (local: Tesseract fallback)

`apps/document-service/src/services/ocr.ts`:
```typescript
import axios from 'axios';

// For local dev: use AWS Textract (requires AWS keys in .env)
// Fallback: Tesseract via tesseract.js npm package

export async function runOCR(filePath: string): Promise<string> {
  const ocrProvider = process.env.OCR_PROVIDER ?? 'local_tesseract';

  if (ocrProvider === 'local_tesseract') {
    return runTesseractLocal(filePath);
  }

  if (ocrProvider === 'textract') {
    return runAWSTextract(filePath);
  }

  throw new Error(`Unknown OCR provider: ${ocrProvider}`);
}

async function runTesseractLocal(filePath: string): Promise<string> {
  // npm install tesseract.js
  const { createWorker } = await import('tesseract.js');
  const worker = await createWorker('eng');
  const { data } = await worker.recognize(filePath);
  await worker.terminate();
  return data.text;
}

async function runAWSTextract(filePath: string): Promise<string> {
  const fs = await import('fs');
  const { TextractClient, DetectDocumentTextCommand } = await import('@aws-sdk/client-textract');
  const client = new TextractClient({ region: process.env.AWS_REGION });

  const fileBytes = fs.readFileSync(filePath);
  const response = await client.send(new DetectDocumentTextCommand({
    Document: { Bytes: fileBytes },
  }));

  return (response.Blocks ?? [])
    .filter((b) => b.BlockType === 'LINE' && b.Text)
    .map((b) => b.Text!)
    .join('\n');
}
```

```typescript
import axios from 'axios';

export async function normalizeEntities(entities: any[]): Promise<any[]> {
  return Promise.all(entities.map(async (entity) => {
    if (entity.label === 'CHEMICAL') {
      // RxNorm lookup for drugs
      const rxNormCode = await lookupRxNorm(entity.text);
      return { ...entity, fieldName: 'medication_name', normalizedValue: entity.text, rxNormCode };
    }
    if (entity.label === 'DISEASE') {
      return { ...entity, fieldName: 'diagnosis', normalizedValue: entity.text };
    }
    // Lab values need pattern matching on surrounding context — basic regex for now
    return { ...entity, fieldName: 'other', normalizedValue: entity.text };
  }));
}

async function lookupRxNorm(drugName: string): Promise<string | null> {
  try {
    const resp = await axios.get(
      `https://rxnav.nlm.nih.gov/REST/rxcui.json?name=${encodeURIComponent(drugName)}&search=1`,
      { timeout: 5000 }
    );
    return resp.data?.idGroup?.rxnormId?.[0] ?? null;
  } catch {
    return null;  // normalize failure is non-fatal
  }
}
```

## Step 2.9 — Confidence Gating + Auto-Commit

`apps/document-service/src/services/confidence.ts`:
```typescript
export function scoreConfidence(entities: any[]) {
  const fields = entities.map((e) => ({
    name: e.fieldName,
    rawValue: e.text,
    normalizedValue: e.normalizedValue ?? e.text,
    confidence: computeScore(e),
  }));
  const avgConfidence = fields.reduce((sum, f) => sum + f.confidence, 0) / (fields.length || 1);
  return { fields, avgConfidence };
}

function computeScore(entity: any): number {
  // Base score from NER model confidence (scispaCy doesn't expose scores — use heuristics)
  let score = 0.7;
  if (entity.rxNormCode) score += 0.15;  // matched in authoritative DB
  if (entity.text.length > 3) score += 0.05;
  return Math.min(score, 1.0);
}
```

**Confidence Tiers:**
| Score | Tier | Action |
|---|---|---|
| ≥ 0.85 | HIGH | Auto-write to Patient 360 |
| 0.60 – 0.84 | MEDIUM | Hold in `ExtractedField`, request user confirmation |
| < 0.60 | LOW | Flag for manual review, do not write |

`apps/document-service/src/services/commit.ts`:
```typescript
import { prisma } from '@medibot/db';

export async function autoCommitHighConfidence(documentId: string) {
  const doc = await prisma.healthDocument.findUnique({ where: { documentId } });
  if (!doc) return;

  const highConfFields = await prisma.extractedField.findMany({
    where: { documentId, confidence: { gte: 0.85 }, status: 'confirmed' },
    include: { provenance: true },
  });

  for (const field of highConfFields) {
    await commitFieldToPatient360(doc.patientId, field);
  }
}

export async function commitConfirmedFields(documentId: string) {
  const doc = await prisma.healthDocument.findUnique({ where: { documentId } });
  if (!doc) return;

  const confirmedFields = await prisma.extractedField.findMany({
    where: { documentId, status: 'confirmed' },
    include: { provenance: true },
  });

  for (const field of confirmedFields) {
    await commitFieldToPatient360(doc.patientId, field);
  }
}

async function commitFieldToPatient360(patientId: string, field: any) {
  const provenanceId = field.provenanceId;

  if (field.fieldName === 'medication_name') {
    await prisma.medication.create({
      data: { patientId, name: field.normalizedValue ?? field.rawValue, provenanceId },
    });
    await prisma.healthEvent.create({
      data: {
        patientId,
        eventType: 'prescription',
        eventTime: new Date(),
        summary: `Medication extracted from document: ${field.normalizedValue ?? field.rawValue}`,
        provenanceId,
      },
    });
  }

  if (field.fieldName === 'diagnosis') {
    await prisma.medicalCondition.create({
      data: { patientId, name: field.normalizedValue ?? field.rawValue, provenanceId },
    });
  }
  // Add more field types: lab_value → LabResult, allergy → Allergy, etc.
}
```
